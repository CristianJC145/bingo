-- Verificación: deben retornar 13 usuarios, 13 bloques y 52.000 registros.
SELECT COUNT(*) AS usuarios_demo
FROM users WHERE username LIKE 'reg_demo_%';

SELECT COUNT(*) AS bloques,
       SUM(end_card - start_card + 1) AS cartones_asignados
FROM card_assignments
WHERE start_card BETWEEN 1 AND 52000;

SELECT COUNT(*) AS cartones_registrados,
       COUNT(DISTINCT municipio) AS municipios,
       SUM(discapacidad = 'SI') AS con_discapacidad,
       SUM(cuidador = 'SI') AS cuidadores,
       SUM(adulto_mayor = 'SI') AS adultos_mayores
FROM card_registrations
WHERE card_number BETWEEN 1 AND 52000;

SELECT
  a.municipio,
  u.username,
  a.start_card,
  a.end_card,
  COUNT(r.card_number) AS registrados
FROM card_assignments a
JOIN users u ON u.id = a.user_id
LEFT JOIN card_registrations r ON r.card_number BETWEEN a.start_card AND a.end_card
WHERE a.start_card BETWEEN 1 AND 52000
GROUP BY a.id, a.municipio, u.username, a.start_card, a.end_card
ORDER BY a.start_card;
